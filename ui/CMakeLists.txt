cmake_minimum_required(VERSION 3.20)
project(uicfg CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UICFG_NLOHMANN_INCLUDE_DIR "" CACHE PATH "header-only nlohmann/json include dir (use when cross-compiling)")
if(NOT UICFG_NLOHMANN_INCLUDE_DIR AND NOT CMAKE_CROSSCOMPILING)
  find_package(nlohmann_json 3.11 REQUIRED)
endif()
add_library(uicfg src/validate.cpp src/store.cpp src/workspace.cpp src/settings.cpp src/layout_ops.cpp src/link_policy.cpp src/close_confirm.cpp src/tab_model.cpp src/session_tree.cpp src/reconnect.cpp src/shell_logic.cpp src/shell_models.cpp)
target_include_directories(uicfg PUBLIC include)
target_include_directories(uicfg PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../app/common)
if(UICFG_NLOHMANN_INCLUDE_DIR)
  target_include_directories(uicfg SYSTEM PUBLIC ${UICFG_NLOHMANN_INCLUDE_DIR})
elseif(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "cross-compiling uicfg: set UICFG_NLOHMANN_INCLUDE_DIR to a dir containing nlohmann/json.hpp (not /usr/include)")
else()
  target_link_libraries(uicfg PUBLIC nlohmann_json::nlohmann_json)
endif()
if(MSVC)
  target_compile_options(uicfg PRIVATE /utf-8 /W4)
endif()
enable_testing()
foreach(t validate_tests store_tests workspace_tests settings_tests layout_ops_tests link_policy_tests close_confirm_tests tab_model_tests session_tree_tests reconnect_tests shell_logic_tests shell_models_tests)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} uicfg)
  add_test(NAME uicfg_${t} COMMAND ${t})
endforeach()

# T08 Win32 主外壳（仅 WIN32 / mingw）。在仓库顶层构建时链接 it2w_pane_backend（目标在 app/pane_backend 定义，
# 生成阶段解析，无需顶层改动）；单独 cmake -S ui 时用占位后端，只验证外壳本身。
if(WIN32)
  enable_language(RC)
  add_executable(it2w_shell WIN32 shell_win32/main.cpp shell_win32/it2w_shell.rc)
  target_compile_definitions(it2w_shell PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(it2w_shell PRIVATE uicfg comctl32 comdlg32 shell32 ole32 user32 gdi32)
  if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    target_compile_definitions(it2w_shell PRIVATE IT2W_SHELL_STUB_BACKEND)
  else()
    target_link_libraries(it2w_shell PRIVATE it2w_pane_backend)
  endif()
  if(MINGW)
    target_compile_options(it2w_shell PRIVATE -Wall -Wextra -municode)
    target_link_options(it2w_shell PRIVATE -municode -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_options(it2w_shell PRIVATE /utf-8)
  endif()
endif()
