cmake_minimum_required(VERSION 3.20)
project(uicfg CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json 3.11 REQUIRED)
add_library(uicfg src/validate.cpp src/store.cpp src/workspace.cpp src/settings.cpp src/layout_ops.cpp)
target_include_directories(uicfg PUBLIC include)
target_link_libraries(uicfg PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(uicfg PRIVATE /utf-8 /W4)
endif()
enable_testing()
foreach(t validate_tests store_tests workspace_tests settings_tests layout_ops_tests)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} uicfg)
  add_test(NAME uicfg_${t} COMMAND ${t})
endforeach()
