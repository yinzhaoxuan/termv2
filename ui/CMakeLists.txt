cmake_minimum_required(VERSION 3.20)
project(uicfg CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UICFG_NLOHMANN_INCLUDE_DIR "" CACHE PATH "header-only nlohmann/json include dir (use when cross-compiling)")
if(NOT UICFG_NLOHMANN_INCLUDE_DIR AND NOT CMAKE_CROSSCOMPILING)
  find_package(nlohmann_json 3.11 REQUIRED)
endif()
add_library(uicfg src/validate.cpp src/store.cpp src/workspace.cpp src/settings.cpp src/layout_ops.cpp src/link_policy.cpp)
target_include_directories(uicfg PUBLIC include)
if(UICFG_NLOHMANN_INCLUDE_DIR)
  target_include_directories(uicfg SYSTEM PUBLIC ${UICFG_NLOHMANN_INCLUDE_DIR})
elseif(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "cross-compiling uicfg: set UICFG_NLOHMANN_INCLUDE_DIR to a dir containing nlohmann/json.hpp (not /usr/include)")
else()
  target_link_libraries(uicfg PUBLIC nlohmann_json::nlohmann_json)
endif()
if(MSVC)
  target_compile_options(uicfg PRIVATE /utf-8 /W4)
endif()
enable_testing()
foreach(t validate_tests store_tests workspace_tests settings_tests layout_ops_tests link_policy_tests)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} uicfg)
  add_test(NAME uicfg_${t} COMMAND ${t})
endforeach()
