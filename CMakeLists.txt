# CORE-01/02：每个向量一条 CTest，所有切分方式输出必须一致；另有黄金输出比对。
enable_testing()
set(ITCORE_VECTORS "/workspace/qa/vectors/core01_02.json" CACHE FILEPATH "CORE-01/02 测试向量")
function(itcore_add_vector_tests file golden)
  file(READ "${file}" _vec_json)
  string(JSON _n LENGTH "${_vec_json}")
  math(EXPR _last "${_n} - 1")
  foreach(_i RANGE ${_last})
    string(JSON _case GET "${_vec_json}" ${_i} case)
    string(JSON _name GET "${_vec_json}" ${_i} name)
    add_test(NAME "${_case}.${_name}.splits"
             COMMAND itcore_test_tokens "${file}" --case "${_name}" --check)
  endforeach()
  get_filename_component(_g "${golden}" NAME_WE)
  add_test(NAME "${_g}.golden"
           COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:itcore_test_tokens> -DVEC=${file}
                   -DGOLDEN=${golden} -P ${CMAKE_SOURCE_DIR}/tests/compare_golden.cmake)
endfunction()
if(EXISTS "${ITCORE_VECTORS}")
  itcore_add_vector_tests("${ITCORE_VECTORS}" "${CMAKE_SOURCE_DIR}/tests/golden/core01_02.tokens")
else()
  message(WARNING "未找到测试向量 ${ITCORE_VECTORS}，跳过 CORE-01/02 CTest")
endif()
itcore_add_vector_tests("${CMAKE_SOURCE_DIR}/tests/vectors/extra.json" "${CMAKE_SOURCE_DIR}/tests/golden/extra.tokens")
