# 测试目标登记（另可对未注册 add_test 的测试程序设 IT2W_TEST=ON）：覆盖 add_test，把 COMMAND 的第一个参数（若是目标）记入全局属性 IT2W_TEST_TARGETS，
# 文件末尾据此决定 NDEBUG（不再按目标名判断）。所有子目录的 add_test 都经过这里。
function(add_test)
  _add_test(${ARGV})
  cmake_parse_arguments(_at "" "NAME" "COMMAND" ${ARGV})
  if(_at_NAME)
    list(GET _at_COMMAND 0 _exe)
  else()
    list(GET ARGV 1 _exe)   # 旧式 add_test(name exe ...)
  endif()
  set_property(GLOBAL APPEND PROPERTY IT2W_TEST_TARGETS_${CMAKE_CURRENT_SOURCE_DIR} "${_exe}")
  set_property(GLOBAL APPEND PROPERTY IT2W_TEST_DIRS "${CMAKE_CURRENT_SOURCE_DIR}")
endfunction()

function(itcore_add_vector_tests file golden)
  file(READ "${file}" _vec_json)
  string(JSON _n LENGTH "${_vec_json}")
  math(EXPR _last "${_n} - 1")
  foreach(_i RANGE ${_last})
    string(JSON _case GET "${_vec_json}" ${_i} case)
    string(JSON _name GET "${_vec_json}" ${_i} name)
    add_test(NAME "${_case}.${_name}.splits"
             COMMAND itcore_test_tokens "${file}" --case "${_name}" --check)
  endforeach()
  get_filename_component(_g "${golden}" NAME_WE)
  add_test(NAME "${_g}.golden"
           COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:itcore_test_tokens> -DVEC=${file}
                   -DGOLDEN=${golden} -P ${CMAKE_SOURCE_DIR}/tests/compare_golden.cmake)
endfunction()

# NDEBUG 按目标决定：全局 *_FLAGS_<CONFIG> 已去掉 -DNDEBUG（见文件头）；
# 非测试目标在优化配置下定义 NDEBUG，测试目标不定义（assert 在 Release/RelWithDebInfo 真机包里也生效）。
# 不用 -UNDEBUG：它依赖命令行顺序（-DNDEBUG 在 CMAKE_CXX_FLAGS_<CONFIG> 里，可能排在其后），不可靠。
function(it2w_collect_targets dir out)
  get_property(_t DIRECTORY ${dir} PROPERTY BUILDSYSTEM_TARGETS)
  get_property(_s DIRECTORY ${dir} PROPERTY SUBDIRECTORIES)
  foreach(_d ${_s})
    it2w_collect_targets(${_d} _sub)
    list(APPEND _t ${_sub})
  endforeach()
  set(${out} ${_t} PARENT_SCOPE)
endfunction()
it2w_collect_targets(${CMAKE_CURRENT_SOURCE_DIR} _all_targets)
# add_test 登记的命令若是本项目目标，即为测试目标（属性 IT2W_TEST=ON）
get_property(_tdirs GLOBAL PROPERTY IT2W_TEST_DIRS)
list(REMOVE_DUPLICATES _tdirs)
set(_test_targets "")
foreach(_d ${_tdirs})
  get_property(_names GLOBAL PROPERTY IT2W_TEST_TARGETS_${_d})
  foreach(_n ${_names})
    if(TARGET ${_n})
      list(APPEND _test_targets ${_n})
      set_property(TARGET ${_n} PROPERTY IT2W_TEST ON)
    endif()
  endforeach()
endforeach()
list(REMOVE_DUPLICATES _test_targets)
foreach(_tg ${_all_targets})
  get_target_property(_ty ${_tg} TYPE)
  if(_ty STREQUAL "INTERFACE_LIBRARY" OR _ty STREQUAL "UTILITY")
    continue()
  endif()
  get_target_property(_ist ${_tg} IT2W_TEST)
  if(_ist)
    list(APPEND _test_targets ${_tg})
  else()
    target_compile_definitions(${_tg} PRIVATE $<$<CONFIG:Release,RelWithDebInfo,MinSizeRel>:NDEBUG>)
  endif()
endforeach()
list(REMOVE_DUPLICATES _test_targets)

  foreach(_p COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(_v ${_tg} ${_p})
    if(_v AND "${_v}" MATCHES "NDEBUG")
      message(FATAL_ERROR "测试目标 ${_tg} 的 ${_p} 带 NDEBUG：${_v}")
    endif()
  endforeach()
endforeach()
